/// SwiftData数据管理容器接口
/// 负责：数据模型注册、容器初始化、迁移管理
public protocol SwiftDataContainerProtocol: Actor {
    
    /// 初始化数据容器
    /// - Parameters:
    ///   - models: 需要注册的模型类型
    ///   - configuration: 容器配置
    /// - Returns: 配置好的ModelContainer
    func initialize(
        models: [any PersistentModel.Type],
        configuration: ModelConfiguration
    ) async throws -> ModelContainer
    
    /// 执行数据迁移
    /// - Parameter fromVersion: 源版本号
    func migrate(fromVersion: String) async throws
    
    /// 验证数据完整性
    /// - Returns: 验证报告
    func validateDataIntegrity() async throws -> DataIntegrityReport
    
    /// 容器状态变化事件流
    var containerStateChanged: AsyncStream<ContainerState> { get }
    
    /// 错误定义
    enum ContainerError: ModuleError {
        case initializationFailed(reason: String)
        case migrationFailed(fromVersion: String, toVersion: String)
        case integrityCheckFailed(details: [String])
    }
    
    /// 容器状态
    enum ContainerState: Sendable {
        case uninitialized
        case initializing
        case ready
        case migrating
        case error(Error)
    }
}

/// 图片压缩与格式转换接口
/// 负责：图片压缩、HEIC/HEIF转换、存储优化
public protocol ImageCompressorProtocol: Actor {
    
    /// 压缩图片
    /// - Parameters:
    ///   - image: 原始图片
    ///   - quality: 压缩质量 (0.0-1.0)
    ///   - maxSize: 最大尺寸限制
    /// - Returns: 压缩后的图片数据
    func compress(
        image: UIImage,
        quality: Float,
        maxSize: CGSize?
    ) async throws -> Data
    
    /// 批量压缩
    /// - Parameters:
    ///   - images: 图片数组
    ///   - options: 压缩选项
    /// - Returns: 压缩结果流
    func batchCompress(
        images: [UIImage],
        options: CompressionOptions
    ) -> AsyncThrowingStream<CompressionResult, Error>
    
    /// HEIC/HEIF格式转换
    /// - Parameters:
    ///   - data: 原始数据
    ///   - targetFormat: 目标格式
    func convertFormat(
        data: Data,
        to targetFormat: ImageFormat
    ) async throws -> Data
    
    /// 获取压缩统计信息
    func getCompressionStats() async -> CompressionStatistics
    
    /// 压缩选项
    struct CompressionOptions: Sendable {
        let quality: Float
        let maxDimension: Int?
        let preserveMetadata: Bool
        let format: ImageFormat
    }
    
    /// 图片格式
    enum ImageFormat: String, Sendable {
        case jpeg, heic, png, webp
    }
    
    /// 压缩结果
    struct CompressionResult: Sendable {
        let originalSize: Int
        let compressedSize: Int
        let compressionRatio: Float
        let data: Data
    }
}

/// CloudKit同步引擎接口
/// 负责：自动同步、冲突检测、合并策略
public protocol SyncEngineProtocol: Actor {
    
    /// 启动同步
    /// - Parameter mode: 同步模式

        case lastWriteWins

        let isActive: Bool
        let lastSyncDate: Date?
        let pendingChanges: Int
        let conflicts: [SyncConflict]
    }
}

// MARK: ErrorHandler
// ==========================================

/// 全局错误处理器接口
/// 负责：错误捕获、恢复策略、用户提示
public protocol ErrorHandlerProtocol: Actor {
    
    /// 处理错误
    func handle(
        _ error: Error,
        context: ErrorContext,
        recovery: RecoveryStrategy?
    ) async -> ErrorResolution
    
    /// 注册错误处理器
    func registerHandler(
        for errorType: any Error.Type,
        handler: @escaping ErrorHandlerClosure
    )
    
    /// 获取错误历史
    func getErrorHistory(
        limit: Int
    ) async -> [ErrorRecord]
    
    /// 清除错误历史
    func clearErrorHistory() async
    
    /// 错误通知流
    var errorNotifications: AsyncStream<ErrorNotification> { get }
    
    /// 错误上下文
    struct ErrorContext: Sendable {
        let module: String
        let operation: String
        let metadata: [String: Any]
        let timestamp: Date
    }
    
    /// 恢复策略
    enum RecoveryStrategy: Sendable {
        case retry(maxAttempts: Int, delay: TimeInterval)
        case fallback(handler: @Sendable () async -> Void)
        case ignore
        case escalate
    }
    
    /// 错误解决结果
    enum ErrorResolution: Sendable {
        case recovered
        case failed(Error)
        case userCancelled
        case escalated
    }

// MARK: - Module Integration Guidelines

/*
 模块集成指南：
 
 1. 依赖注入：
    - 所有模块通过协议注入，不直接依赖具体实现
    - 使用 ModuleCoordinator 管理模块生命周期
 
 2. 错误处理：
    - 每个模块定义自己的错误类型，遵循 ModuleError 协议
    - 提供用户友好的错误消息
 
 3. 并发安全：
    - 所有模块使用 Actor 隔离
    - 公共数据结构必须是 Sendable
 
 4. 事件通信：
    - 使用 AsyncStream 进行异步事件传递
    - 避免直接回调，使用订阅模式
 
 5. 性能考虑：
    - 重操作使用 Task.detached
    - 提供进度反馈的 AsyncStream
 
 6. 测试策略：
    - 每个协议提供 Mock 实现
    - 使用依赖注入便于测试
 
 7. 版本管理：
    - 使用语义化版本控制
    - 接口变更需要版本升级
 */
